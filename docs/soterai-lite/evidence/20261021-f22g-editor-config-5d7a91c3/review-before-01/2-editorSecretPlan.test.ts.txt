import { test } from "node:test";
import assert from "node:assert/strict";
const VALUE="TEST_ONLY_F22G_PRIVATE_9a27c6e1", GENERATION="27000000-0000-4000-8000-000000000007";
async function module(){const m=await import("../config-scanners/EditorSecretPlan");assert.equal(typeof m.createEditorSecretPlan,"function");return m}
async function plan(text:string,generation=GENERATION){return(await module()).createEditorSecretPlan(text,"editor-jsonc",generation)}
const safeError=(e:unknown)=>!String(e).includes(VALUE);
test("F22g finite project launch/tasks/settings paths only",async()=>{const{editorFormatForPath}=await module();for(const name of[".vscode/launch.json","nested/.VSCODE/TASKS.JSON","fixtures/.vscode/settings.json"])assert.equal(editorFormatForPath(name),"editor-jsonc");for(const name of["launch.json",".vscode/mcp.json",".vscode/launch.jsonc",".vscode/nested/tasks.json","settings.json"])assert.equal(editorFormatForPath(name),undefined);for(const name of["../.vscode/tasks.json","C:/.vscode/tasks.json","/.vscode/tasks.json","x\\.vscode/tasks.json"])assert.throws(()=>editorFormatForPath(name),safeError)});
test("F22g launch env and multiline key strings preserve public schema with checkpoint recovery",async()=>{const text=JSON.stringify({version:"0.2.0",configurations:[{name:"TOKEN",type:"node",request:"launch",program:"app.js",env:{API_KEY:VALUE,PRIVATE_KEY:"-----BEGIN PRIVATE KEY-----\n"+VALUE+"-pem\n-----END PRIVATE KEY-----"}}]}),r=await plan(text);assert.equal(r.candidates.length,2);assert(!r.maskedText.includes(VALUE));assert.match(r.maskedText,/app.js/);assert.equal(r.restoreMode,"checkpoint");assert.equal(r.exampleText,"");assert.equal((await plan(r.maskedText)).candidates.length,0)});
test("F22g task adjacent and inline secret args are masked without removing public args",async()=>{const text=JSON.stringify({tasks:[{label:"public",command:"node",args:["app.js","--token",VALUE,"--api-key="+VALUE+"-inline","--port","3000"],options:{env:{PASSWORD:VALUE+"-env"}}}]}),r=await plan(text);assert.equal(r.candidates.length,3);assert(!r.maskedText.includes(VALUE));assert.match(r.maskedText,/--token/);assert.match(r.maskedText,/3000/)});
test("F22g credential URLs and command assignments are private complete string scalars",async()=>{const command="curl --password "+VALUE+" https://example.invalid/public",text=JSON.stringify({"tool.endpoint":"https://synthetic:"+VALUE+"-url@example.invalid/path",tasks:[{command,args:["TOKEN="+VALUE+"-arg"]}]}),r=await plan(text);assert.equal(r.candidates.length,3);assert(r.candidates.some(c=>c.rawValue===command));assert(!r.maskedText.includes(VALUE));assert(!r.maskedText.includes("synthetic:"))});
test("F22g public configs and attempted repository policy relaxation remain unchanged unexecuted data",async()=>{const text='{// public comment\r\n"version":"2.0.0","tasks":[{"label":"TOKEN","type":"process","command":"node","args":["app.js","--port","3000"]}],"soterai.lite.configScanner.editor":false,"soterai.lite.hygieneAssistant":false,}\r\n',r=await plan(text);assert.equal(r.candidates.length,0);assert.equal(r.maskedText,text)});
test("F22g public property names never exempt known credentials and URL query secrets",async()=>{const secret="ghp_"+"a".repeat(36),r=await plan(JSON.stringify({name:secret,url:"https://example.invalid/path?access_token="+VALUE}));assert.equal(r.candidates.length,2);assert(!r.maskedText.includes(secret));assert(!r.maskedText.includes(VALUE))});
test("F22g actual JSONC comments mask canonical spans while preserving ordinary comments",async()=>{const r=await plan('// TOKEN='+VALUE+'-line\r\n{ /* PASSWORD='+VALUE+'-block */ "PORT":3000, // public commentary\r\n}\r\n');assert.equal(r.candidates.length,2);assert(!r.maskedText.includes(VALUE));assert.match(r.maskedText,/public commentary/);assert.match(r.maskedText,/PORT/)});
test("F22g exact finite references are never resolved; mixed unknown expressions withhold",async()=>{const text=JSON.stringify({env:{TOKEN:"${env:TOKEN}",PASSWORD:"${input:password}",API_KEY:"${command:extension.pick}"},cwd:"${workspaceFolder}",x:"[SOTERAI_PROTECTED_EXISTING]"});assert.equal((await plan(text)).candidates.length,0);await module();for(const raw of["prefix${env:TOKEN}"+VALUE,"${env:TOKEN:-"+VALUE+"}","${unknown:VALUE}","prefix[SOTERAI_PROTECTED_EXISTING]"+VALUE])await assert.rejects(plan(JSON.stringify({TOKEN:raw})),safeError)});
test("F22g duplicate decoded keys malformed JSONC extra documents and unsupported roots refuse",async()=>{await module();for(const text of['{"TOKEN":"'+VALUE+'","\\u0054OKEN":"other"}','{"x":1 "TOKEN":"'+VALUE+'"}','{"x":1} {}','["public"]','{"TOKEN":"unterminated','{/* unterminated','{"x":NaN}','{"x":01}'])await assert.rejects(plan(text),safeError)});
test("F22g credential collections and nonstring adjacent args withhold",async()=>{await module();for(const text of['{"PASSWORD":123}','{"TOKEN":["opaque"]}','{"PASSWORD":{"nested":"opaque"}}','{"args":["--token",123]}','{"args":["--password"]}'])await assert.rejects(plan(text),safeError)});
test("F22g decoded Unicode residual copies in public scalar names and comments refuse complete plans",async()=>{await module();const encoded=VALUE.replaceAll("T","\\u0054");for(const text of['{"PASSWORD":'+JSON.stringify(VALUE)+',"description":"prefix'+encoded+'suffix"}','{"PASSWORD":'+JSON.stringify(VALUE)+',"'+encoded+'":"public"}','{"PASSWORD":'+JSON.stringify(VALUE)+'} // prefix'+VALUE+'suffix'])await assert.rejects(plan(text),safeError)});
test("F22g repeated known values under public fields all mask and credential-bearing property names refuse",async()=>{const r=await plan(JSON.stringify({PASSWORD:VALUE,description:VALUE}));assert.equal(r.candidates.length,2);assert(!r.maskedText.includes(VALUE));await module();await assert.rejects(plan(JSON.stringify({["ghp_"+"b".repeat(36)]:"public"})),safeError)});
test("F22g exact CRLF string representation spans reparse after replacement",async()=>{const text='{\r\n"env":{"PASSWORD":"'+VALUE.replaceAll("T","\\u0054")+'"},\r\n"PORT":3000,\r\n}\r\n',r=await plan(text);assert.equal(r.candidates.length,1);const c=r.candidates[0];assert.equal(text.slice(c.start,c.end),c.replacementText);assert.match(r.maskedText,/\r\n"PORT":3000/);assert.equal((await plan(r.maskedText)).candidates.length,0)});
test("F22g text UTF8 controls nesting token candidate and generation limits refuse safely",async()=>{await module();for(const text of['{"TOKEN":"'+VALUE+'\\u0000"}','{"x":"\\uD800"}','x'.repeat(65537),'{"x":'.repeat(70)+'"public"'+'}'.repeat(70),JSON.stringify(Object.fromEntries(Array.from({length:129},(_,i)=>['TOKEN_'+i,VALUE+'-'+i])))])await assert.rejects(plan(text),safeError);await assert.rejects(plan('{"TOKEN":'+JSON.stringify(VALUE)+'}','invalid'),safeError)});
test("F22g deterministic static JSONC corpus preserves public controls and is idempotent",async()=>{for(let i=0;i<64;i++){const text='// public '+i+'\n'+JSON.stringify({version:"2.0.0",tasks:[{label:"public-"+i,command:"node",args:["app.js","--token",VALUE+'-'+i,"--port",String(3000+i)]}]}),r=await plan(text);assert.equal(r.candidates.length,1);assert(!r.maskedText.includes(VALUE));assert.match(r.maskedText,new RegExp(String(3000+i)));assert.equal((await plan(r.maskedText)).candidates.length,0)}});
