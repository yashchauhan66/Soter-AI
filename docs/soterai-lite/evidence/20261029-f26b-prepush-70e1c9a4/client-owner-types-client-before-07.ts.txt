// Real Ed25519, checked ordinary artifacts and physical directories. Injected
// request seams are explicit; separate tests use the actual numeric-loopback
// transport. No child, Git push, normal profile, secrets, signals or deletion.
const options = { skip: process.platform !== "win32", timeout: 12000 };
const TRANSCRIPT = Buffer.from("refs/heads/main " + "1".repeat(40) + " refs/heads/main " + "0".repeat(40) + "\n");
const PRIVATE = "TEST_ONLY_F26B_CLIENT_WORKTREE_7526b18c";
const digest = (raw: Buffer) => createHash("sha256").update(raw).digest("hex");
interface Context { endpoint: string; request: Readonly<HookRequest>; raw: Buffer; deadline: number;
    reply(patch?: Partial<HookDecision>, key?: KeyObject): Buffer }
interface Settings { pin?: Buffer; transcript?: Buffer; runtime?: string; worker?: string; now?: () => number; deadline?: number;
    request?: (context: Context) => Promise<Buffer | undefined> }
async function fixture() {
    const root = await mkdtemp(join(tmpdir(), "soter-f26b-client-"));
    for (const name of ["assets", "profile", "synthetic-windows"]) await mkdir(join(root, name));
    const worker = join(root, "assets", "pre-push-client.js"), runtime = join(root, "assets", "Code.exe"), descriptor = join(root, "profile", "endpoint.json");
    const workerBytes = Buffer.from("// Synthetic prepush worker, never executed.\n"), runtimeBytes = Buffer.from("MZ_SYNTHETIC_F26B_RUNTIME_NEVER_EXECUTED");
    await writeFile(worker, workerBytes, { flag: "wx" }); await writeFile(runtime, runtimeBytes, { flag: "wx" });
    const key = generateKeyPairSync("ed25519").privateKey, generation = randomUUID(), session = randomUUID();
    const publicKey = (createPublicKey(key).export({ format: "der", type: "spki" }) as Buffer).toString("base64");
    const pin: HookClientPin = { version: 1, root, generation, publicKey, worker, workerSha256: digest(workerBytes), runtime,
        runtimeSha256: digest(runtimeBytes), descriptor, systemRoot: join(root, "synthetic-windows") };
    const signedDescriptor = (patch: Partial<HookDescriptor> = {}, signer = key) => {
        const raw = signHookDescriptor({ version: 1, root, generation, session, endpoint: "http://127.0.0.1:43127/pre-push",
            workerSha256: pin.workerSha256, runtimeSha256: pin.runtimeSha256, ...patch }, signer); assert(raw); return raw;
    };
    await writeFile(descriptor, signedDescriptor(), { flag: "wx" });
    const requests: Context[] = [];
    function context(endpoint: string, raw: Buffer, deadline: number, transcript = TRANSCRIPT): Context {
        const request = parseHookRequest(raw, { generation, session }); assert(request, "Client must send canonical bound push request metadata.");
        const identity = pushRootIdentity(root), transcriptSha256 = prePushTranscriptDigest(transcript); assert(identity && transcriptSha256);
        return { endpoint, request, raw: Buffer.from(raw), deadline, reply(patch = {}, signer = key) {
            const bytes = signHookDecision({ version: 1, root, generation, session, nonce: request.nonce, status: "ALLOW",
                transcriptSha256, rootIdentity: identity, ...patch }, signer); assert(bytes); return bytes;
        } };
    }
    return { root, worker, runtime, descriptor, key, generation, session, pin, requests, signedDescriptor, context,
        rawPin: () => Buffer.from(JSON.stringify(pin)),
        run(settings: Settings = {}) {
            const transcript = settings.transcript ?? TRANSCRIPT;
            return runPrePushClient(settings.pin ?? Buffer.from(JSON.stringify(pin)), transcript, {
                runtime: settings.runtime ?? runtime, worker: settings.worker ?? worker, now: settings.now ?? (() => 1000), deadline: settings.deadline,
                request: async (endpoint, raw, deadline) => { const value = context(endpoint, raw, deadline, transcript); requests.push(value);
                    return settings.request ? settings.request(value) : value.reply(); }
            });
        } };
}
type Fixture = Awaited<ReturnType<typeof fixture>>;
async function unavailable(f: Fixture, settings: Settings = {}) { assert.equal(await f.run(settings), "UNAVAILABLE"); }
async function replaceIdentical(file: string) {
    const before = await lstat(file), bytes = await readFile(file), incoming = file + ".incoming-" + randomUUID();
    await writeFile(file + ".retained-" + randomUUID(), bytes, { flag: "wx" }); await writeFile(incoming, bytes, { flag: "wx" }); await rename(incoming, file);
    assert.notEqual((await lstat(file)).ino, before.ino, "Identical replacement must change physical file identity.");
}
async function transportFixture(t: TestContext, response?: (context: Context, req: IncomingMessage, res: ServerResponse) => void) {
    const f = await fixture(), paths: string[] = [];
    const server = createServer({ maxHeaderSize: 4096 }, (req, res) => {
        paths.push(req.url ?? ""); const chunks: Buffer[] = [];
        req.on("data", chunk => chunks.push(Buffer.from(chunk))); req.once("end", () => {
            const context = f.context("http://127.0.0.1:1/pre-push", Buffer.concat(chunks), performance.now() + 60000);
            if (response) { response(context, req, res); return; }
            const bytes = context.reply(); res.writeHead(200, { "Content-Type": "application/json", "Content-Length": String(bytes.length), "Connection": "close" }); res.end(bytes);
        });
    });
    server.on("clientError", (_error, socket) => socket.end());
    await new Promise<void>((resolve, reject) => { server.once("error", reject); server.listen(0, "127.0.0.1", resolve); });
    t.after(async () => { await new Promise<void>(resolve => server.close(() => resolve())); });
    const address = server.address(); assert(address && typeof address === "object" && address.address === "127.0.0.1");
    await writeFile(f.descriptor, f.signedDescriptor({ endpoint: "http://127.0.0.1:" + address.port + "/pre-push" }));
    return { ...f, paths, actual: () => runPrePushClient(f.rawPin(), TRANSCRIPT, { runtime: f.runtime, worker: f.worker }) };
}

test("F26b client actual numeric-loopback transport uses the signed pre-push route", options, async t => {
    const f = await transportFixture(t); assert.equal(await f.actual(), "ALLOW"); assert.deepEqual(f.paths, ["/pre-push"]);
});
test("F26b client unchanged genuine artifacts and signed exact transcript remain useful", options, async () => {
    const f = await fixture(), pin = f.rawPin(), raw = Buffer.from(TRANSCRIPT), before = new Map<string, Buffer>();
    for (const file of [f.worker, f.runtime, f.descriptor]) before.set(file, await readFile(file));
    assert(parseHookClientPin(pin)); assert.equal(await f.run({ pin, transcript: raw }), "ALLOW");
    assert(pin.equals(f.rawPin()) && raw.equals(TRANSCRIPT)); assert.equal(f.requests.length, 1);
    assert.equal(f.requests[0].endpoint, "http://127.0.0.1:43127/pre-push"); assert(f.requests[0].deadline > 1000 && f.requests[0].deadline <= 61000);
    for (const [file, bytes] of before) assert((await readFile(file)).equals(bytes));
});

test("F26b client nonce freshness and request privacy exclude worktree and ambient bytes", options, async () => {
    const f = await fixture(); await writeFile(join(f.root, "private-worktree.txt"), PRIVATE, { flag: "wx" });
    assert.equal(await f.run(), "ALLOW"); assert.equal(await f.run(), "ALLOW");
    assert.notEqual(f.requests[0].request.nonce, f.requests[1].request.nonce);
    for (const request of f.requests) {
        assert.equal(Object.keys(request.request).sort().join(","), "generation,nonce,session,transcript,version");
        assert(!request.raw.includes(Buffer.from(PRIVATE)) && !request.raw.includes(Buffer.from("private-worktree.txt")) && !request.raw.includes(Buffer.from("NODE_OPTIONS")));
    }
});

test("F26b client inherited deadline shortens whole-operation authority and cannot extend sixty seconds", options, async () => {
    const f = await fixture(); let current = 1000;
    assert.equal(await f.run({ now: () => current, deadline: 1500, request: async value => { assert.equal(value.deadline, 1500); current = 1499; return value.reply(); } }), "ALLOW");
    current = 1000; await unavailable(f, { now: () => current, deadline: 1500, request: async value => { current = 1500; return value.reply(); } });
    assert.equal(await f.run({ deadline: 999999, request: async value => { assert.equal(value.deadline, 61000); return value.reply(); } }), "ALLOW");
});
test("F26b client actual transport rejects wrong status content type duplicate and chunked response framing", options, async t => {
    for (const kind of ["status", "type", "duplicate", "chunked"] as const) {
        const f = await transportFixture(t, (value, _req, res) => {
            const raw = value.reply();
            if (kind === "duplicate") { res.writeHead(200, ["Content-Type", "application/json", "Content-Type", "application/json", "Content-Length", String(raw.length), "Connection", "close"]); }
            else res.writeHead(kind === "status" ? 503 : 200, { "Content-Type": kind === "type" ? "text/plain" : "application/json",
                ...(kind === "chunked" ? {} : { "Content-Length": String(raw.length) }), "Connection": "close" });
            res.end(raw);
        }); assert.equal(await f.actual(), "UNAVAILABLE"); assert.deepEqual(f.paths, ["/pre-push"]);
    }
});
