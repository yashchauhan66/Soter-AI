export function beforeActivate(context: vscode.ExtensionContext): void {
    assert.match(resolve(process.env.USERPROFILE ?? ""), /^D:\\SoterAI-Lite-test-profiles\\/i); assert.match(resolve(vscode.workspace.workspaceFolders![0].uri.fsPath), /^D:\\SoterAI-Lite-test-profiles\\/i); selected = source();
    listeners.push(vscode.workspace.onDidChangeConfiguration(e => note(e.affectsConfiguration("soterai") ? "soterai-configuration-event" : "other-configuration-event")), vscode.extensions.onDidChange(() => note("installed-extension-metadata-event")), context.secrets.onDidChange(e => { if (e.key === KEY) note("hygiene-key-event"); if (e.key === "soterai.vaultKey") note("vault-key-event"); }));
    const update = context.workspaceState.update; patch(context.workspaceState, "update", async (key: string, value: unknown) => { const tag = key === "soterai.aiForbidden.v1" ? "saved-policy" : key === "soterai.lite.policyDrift.v1" ? "drift-baseline" : "other-state"; note(tag + "-update-start"); const result = await update.call(context.workspaceState, key, value); note(tag + "-update-end"); return result; });
    const store = context.secrets.store; patch(context.secrets, "store", async (key: string, value: string) => { if (key === KEY || key === "soterai.vaultKey") { stores++; assert.equal(Buffer.from(value, "base64").length, 32); } return store.call(context.secrets, key, value); });
    const open = vscode.workspace.openTextDocument; patch(vscode.workspace, "openTextDocument", (...args: unknown[]) => { if (isSource(args[0])) sourceCalls++; const input = args[0] as { content?: string }; if (input?.content) { noValues(input.content); metadataDocuments++; } return Reflect.apply(open, vscode.workspace, args); }); listeners.push(vscode.workspace.onDidOpenTextDocument(d => { if (isSource(d.uri)) sourceEvents++; }));
    const show = vscode.window.showTextDocument; patch(vscode.window, "showTextDocument", async (...args: unknown[]) => { const d = args[0] as vscode.TextDocument; noValues(d.getText()); displayed.push(d); return Reflect.apply(show, vscode.window, args); });
    patch(vscode.window, "showOpenDialog", async () => cancelFile ? undefined : selected ? [selected] : undefined); patch(vscode.window, "showQuickPick", async (items: readonly unknown[]) => items[0]);
    patch(vscode.window, "showWarningMessage", async (message: string, options: vscode.MessageOptions, ...items: string[]) => {
        noValues([message, options]); const action = items.find(i => ["Analyze saved project files", "Migrate securely", "Restore raw values", "Create metadata export"].includes(i)); if (!action) return undefined; assert.equal(options.modal, true); reviews.push({ action, safe: true });
        if (drift && action === "Analyze saved project files") { drift = false; await writeFile(source().fsPath, 'services: {api: {image: public-image:1}}\n'); }
        if (hold && action === "Analyze saved project files") { entered = true; return new Promise<string | undefined>(r => finish = r); }
        if (disableDisclosure && action === "Restore raw values") { disableDisclosure = false; await vscode.workspace.getConfiguration("soterai").update(FLAG, false, vscode.ConfigurationTarget.Global); }
        return approve ? action : undefined;
    });
    for (const name of ["showInformationMessage", "showErrorMessage"]) patch(vscode.window, name, async (value: unknown) => { noValues(value); return undefined; });
}
