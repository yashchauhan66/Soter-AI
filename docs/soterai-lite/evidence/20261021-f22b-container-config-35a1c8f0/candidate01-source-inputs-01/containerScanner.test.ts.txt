async function fixture(t: TestContext, options: object = {}) {
    const base = await mkdtemp(join(tmpdir(), "soterai-f22b-container-")), root = join(base, "project"), profile = join(base, "profile"); await mkdir(root); await mkdir(profile); stub.resetState(root);
    const context = stub.makeExtensionContext(profile); context.globalStorageUri = stub.Uri.file(profile); initializeBoundaryState(context.workspaceState as never); const file = join(root, "compose.yaml"), uri = stub.Uri.file(file) as unknown as vscode.Uri; await writeFile(file, TEXT); let opens = 0;
    t.mock.method(stub.workspace, "openTextDocument", async () => { opens++; throw Error(VALUE); }); const events = new stub.EventEmitter<{ key: string }>(); context.secrets.onDidChange = events.event as never;
    const controller = new HygieneController(context as never, { ...options, family: "dockerKubernetes" } as never), manager = new VaultManager(context as never);
    t.after(() => { controller.dispose(); events.dispose(); context.subscriptions.forEach(d => d.dispose()); }); return { controller, manager, context, root, profile, file, uri, opens: () => opens, events };
}
