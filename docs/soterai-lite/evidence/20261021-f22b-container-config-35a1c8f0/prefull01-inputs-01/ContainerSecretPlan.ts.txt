export interface ContainerSecretPlan extends PrivateVaultPlan { restoreMode?: "checkpoint" }
const UUID = /^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/i;
const fail = (): never => { throw Error("Container configuration syntax, expression or size is unsupported; existing files are preserved."); };
interface PendingCandidate extends PrivateVaultCandidate { masked: (marker: string) => string }
function relativeValid(relative: string): void {
    if (typeof relative !== "string" || !relative || relative.length > 2048 || /[\\:\x00-\x1f\x7f]/.test(relative) || relative.startsWith("/") || relative.split("/").some(p => !p || p === "." || p === "..")) return fail();
}
export function containerFormatForPath(relative: string): ContainerFormat | undefined {
    relativeValid(relative); const name = relative.split("/").at(-1)!;
    if (/^(?:Dockerfile(?:\.[A-Za-z0-9_-]+)?|[A-Za-z0-9_-]+\.Dockerfile)$/i.test(name)) return "dockerfile";
    return /\.ya?ml$/i.test(name) ? "yaml" : undefined;
}
function wanted(key: string, value: string, force: boolean): boolean {
    return isHygieneCandidate(key, value) || force && !hygieneCredentialName(key) && isHygieneCandidate("SECRET", value);
}
function sourceMap(n: unknown): n is Record<string, unknown> { return !!n && typeof n === "object" && !Array.isArray(n); }
function member(node: unknown, key: string): unknown { return isMap(node) ? node.items.find(pair => isScalar(pair.key) && pair.key.value === key)?.value : undefined; }
/** Fixed YAML grammar, no toJS/reviver/config resolution or code evaluation. */
function yamlDocuments(text: string) {
    const tokens = [...new Parser().parse(text)], pending: Array<{ value: unknown; depth: number }> = tokens.map(value => ({ value, depth: 0 })), comments: Array<{ text: string; offset: number }> = [];
    let work = 0;
    while (pending.length) {
        const { value, depth } = pending.pop()!; if (++work > 32768 || depth > 128) return fail();
        if (Array.isArray(value)) { for (const child of value) pending.push({ value: child, depth: depth + 1 }); continue; }
        if (!sourceMap(value)) continue;
        if (["error", "alias", "anchor", "tag", "directive"].includes(value.type as string)) return fail();
        if (value.type === "comment") { if (typeof value.source !== "string" || typeof value.offset !== "number") return fail(); comments.push({ text: value.source.slice(1), offset: value.offset + 1 }); }
        for (const child of Object.values(value)) if (child && typeof child === "object") pending.push({ value: child, depth: depth + 1 });
    }
    const composer = new Composer({ version: "1.2", schema: "core", strict: true, stringKeys: true, uniqueKeys: true, merge: false, customTags: [], keepSourceTokens: true, prettyErrors: false, logLevel: "silent" });
    const docs = [...composer.compose(tokens, true, text.length)], stream = composer.streamInfo();
    if (!docs.length || docs.length > 16 || stream.errors.length || stream.warnings.length || docs.some(d => d.errors.length || d.warnings.length)) return fail();
    for (const doc of docs) {
        if (!isMap(doc.contents)) return fail(); const services = member(doc.contents, "services"), apiVersion = member(doc.contents, "apiVersion"), kind = member(doc.contents, "kind");
        if (isMap(services)) { if (docs.length !== 1) return fail(); }
        else if (!isScalar(apiVersion) || typeof apiVersion.value !== "string" || !/^(?:[A-Za-z0-9.-]+\/)?v[0-9]+(?:(?:alpha|beta)[0-9]+)?$/.test(apiVersion.value)
            || !isScalar(kind) || typeof kind.value !== "string" || !/^[A-Z][A-Za-z0-9]{0,63}$/.test(kind.value)) return fail();
    }
    return { docs, comments };
}
type Add = (key: string, raw: string, start: number, end: number, masked: (marker: string) => string, force?: boolean) => void;
function yamlCandidates(text: string, add: Add): void {
    const { docs, comments } = yamlDocuments(text); let work = 0;
    for (const doc of docs) {
        const root = doc.contents!; if (!isMap(root)) return fail(); const kind = member(root, "kind"), kubernetes = !isMap(member(root, "services")) && isScalar(kind), secret = kubernetes && kind.value === "Secret";
        const stack: Array<{ node: unknown; hint: string; path: string[]; secretMode?: "clear" | "encoded" }> = [{ node: root, hint: "", path: [] }];
        while (stack.length) {
            const frame = stack.pop()!, n = frame.node; if (++work > 16384) return fail(); if (n == null) { if (frame.secretMode) return fail(); continue; }
            if (isAlias(n) || (n as Node).tag || (n as Scalar).anchor) return fail();
            if (isMap(n)) {
                if (frame.secretMode) return fail();
                const siblingName = member(n, "name"), envName = kubernetes && frame.path.at(-1) === "env" && isScalar(siblingName) && typeof siblingName.value === "string" && /^[A-Za-z_][A-Za-z0-9_]{0,127}$/.test(siblingName.value) ? siblingName.value : undefined;
                for (const pair of n.items) {
                    if (!isScalar(pair.key) || typeof pair.key.value !== "string" || !pair.key.value || pair.key.value.length > 1024 || pair.key.value === "<<") return fail();
                    const key = pair.key.value, secretMode = secret && frame.path.length === 0 && (key === "data" || key === "stringData") ? key === "data" ? "encoded" : "clear" : undefined;
                    // Fixed family identity was validated above. apiVersion is
                    // public protocol metadata, despite its API name token.
                    if (kubernetes && frame.path.length === 0 && (key === "apiVersion" || key === "kind")) continue;
                    if (secretMode) {
                        if (!isMap(pair.value)) return fail();
                        for (const entry of pair.value.items) { if (!isScalar(entry.key) || typeof entry.key.value !== "string" || entry.key.value.length > 1024) return fail(); stack.push({ node: entry.value, hint: entry.key.value, path: [key, entry.key.value], secretMode }); }
                    } else stack.push({ node: pair.value, hint: key === "value" && envName ? envName : key, path: [...frame.path, key] });
                }
            } else if (isSeq(n)) { if (frame.secretMode) return fail(); for (const item of n.items) stack.push({ node: item, hint: frame.hint, path: frame.path }); }
            else if (isScalar(n)) {
                if (!n.range || frame.secretMode && typeof n.value !== "string") return fail();
                if (!["string", "number", "boolean"].includes(typeof n.value)) continue;
                let raw = String(n.value), key = frame.hint, encoded = false, assignment: string | undefined;
                if (frame.secretMode === "encoded") {
                    if (!/^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/.test(raw)) return fail();
                    const bytes = Buffer.from(raw, "base64"), decoded = bytes.toString("utf8"); if (bytes.toString("base64") !== raw || !Buffer.from(decoded, "utf8").equals(bytes)) return fail(); raw = decoded; encoded = true;
                } else if (!kubernetes && frame.path.at(-1) === "environment") {
                    const m = /^([A-Za-z_][A-Za-z0-9_]{0,127})=([\s\S]*)$/.exec(raw);
                    if (m) { key = m[1]; raw = m[2]; assignment = key; }
                    else if (typeof n.value === "string" && /^[A-Za-z_][A-Za-z0-9_]{0,127}$/.test(raw)) continue;
                }
                new SecureEditorModel(raw, false);
                if (wanted(key, raw, !!frame.secretMode) && raw.includes("$")) return fail();
                const start = n.range[0], end = n.range[1], ending = /(?:\r?\n)+$/.exec(text.slice(start, end))?.[0] ?? "";
                add(key, raw, start, end, marker => JSON.stringify(encoded ? Buffer.from(marker).toString("base64") : assignment ? assignment + "=" + marker : marker) + ending, !!frame.secretMode);
            } else return fail();
        }
    }
    for (const comment of comments) for (const c of extractVaultCandidates(comment.text)) add(c.key, c.rawValue, comment.offset + c.start, comment.offset + c.end, marker => marker);
}
interface Word { value: string; start: number; end: number }
function words(text: string): Word[] {
    const result: Word[] = []; let i = 0;
    while (i < text.length) {
        while (/[ \t]/.test(text[i] ?? "") && i < text.length) i++; if (i >= text.length) break;
        const start = i; let quote = "", value = "";
        while (i < text.length && (quote || !/[ \t]/.test(text[i]))) {
            const c = text[i++];
            if (c === "\\") { const next = text[i++]; if (!next || !/[ \\'"\t$]/.test(next)) return fail(); value += next; }
            else if (quote) { if (c === quote) quote = ""; else value += c; }
            else if (c === "'" || c === '"') quote = c; else value += c;
        }
        if (quote || result.length >= 256) return fail(); result.push({ value, start, end: i });
    }
    return result;
}
function dockerCandidates(text: string, add: Add): void {
    const known = new Set(["FROM", "ENV", "ARG", "RUN", "COPY", "ADD", "WORKDIR", "USER", "EXPOSE", "VOLUME", "LABEL", "ENTRYPOINT", "CMD", "SHELL", "HEALTHCHECK", "STOPSIGNAL", "ONBUILD", "MAINTAINER"]);
    let offset = 0, logical = "", mapping: number[] = [];
    const instruction = () => {
        if (!logical.trim()) { logical = ""; mapping = []; return; }
        const m = /^\s*([A-Za-z]+)[ \t]+([\s\S]+)$/.exec(logical); if (!m || !known.has(m[1].toUpperCase()) || logical.includes("<<")) return fail();
        const name = m[1].toUpperCase(), bodyStart = logical.length - m[2].length, body = m[2];
        if (name !== "ENV" && name !== "ARG") {
            if (extractVaultCandidates(body).length || [...body.matchAll(/\b([A-Za-z_][A-Za-z0-9_]*)\s*=/g)].some(match => hygieneCredentialName(match[1]))) return fail();
        } else {
            const parsed = words(body); if (!parsed.length || name === "ARG" && parsed.length !== 1) return fail();
            const assignment = (key: string, raw: string, start: number, end: number) => {
                if (!/^[A-Za-z_][A-Za-z0-9_]{0,127}$/.test(key)) return fail();
                const sourceStart = mapping[bodyStart + start], sourceEnd = mapping[bodyStart + end - 1] + 1; if (wanted(key, raw, false) && (raw.includes("$") || /[\r\n]/.test(text.slice(sourceStart, sourceEnd)))) return fail();
                add(key, raw, sourceStart, sourceEnd, marker => marker);
            };
            if (name === "ARG" && !parsed[0].value.includes("=")) { if (!/^[A-Za-z_][A-Za-z0-9_]{0,127}$/.test(parsed[0].value)) return fail(); }
            else if (!parsed[0].value.includes("=")) {
                if (name !== "ENV" || parsed.length !== 2) return fail(); assignment(parsed[0].value, parsed[1].value, parsed[1].start, parsed[1].end);
            } else for (const word of parsed) {
                const value = /^([A-Za-z_][A-Za-z0-9_]{0,127})=([\s\S]*)$/.exec(word.value), source = body.slice(word.start, word.end), rawKey = /^([A-Za-z_][A-Za-z0-9_]{0,127})=/.exec(source);
                if (!value || !rawKey || value[1] !== rawKey[1]) return fail(); assignment(value[1], value[2], word.start + rawKey[0].length, word.end);
            }
        }
        logical = ""; mapping = [];
    };
    for (const physical of text.split(/(?<=\n)/)) {
        const line = physical.replace(/\r?\n$/, ""), comment = /^[ \t]*#/.exec(line);
        if (comment) {
            if (/^[ \t]*#[ \t]*escape\s*=/i.test(line) && !/^[ \t]*#[ \t]*escape\s*=\s*\\\s*$/i.test(line)) return fail();
            const start = comment[0].length; for (const c of extractVaultCandidates(line.slice(start))) add(c.key, c.rawValue, offset + start + c.start, offset + start + c.end, marker => marker);
        } else {
            const continuation = line.endsWith("\\"), body = continuation ? line.slice(0, -1) : line;
            for (let i = 0; i < body.length; i++) { logical += body[i]; mapping.push(offset + i); }
            if (continuation) { logical += " "; mapping.push(offset + line.length - 1); } else instruction();
        }
        offset += physical.length;
    }
    if (logical.trim()) return fail();
}
/** Private plan only; originals and representations never become tool/UI output. */
export function createContainerSecretPlan(text: string, format: ContainerFormat, generation: string): ContainerSecretPlan {
    try {
        if (format !== "dockerfile" && format !== "yaml" || typeof generation !== "string" || !UUID.test(generation)) return fail(); new SecureEditorModel(text, false);
        const pending: PendingCandidate[] = [], add: Add = (key, raw, start, end, masked, force = false) => {
            new SecureEditorModel(raw, false); if (!wanted(key, raw, force)) return;
            if (!Number.isInteger(start) || !Number.isInteger(end) || start < 0 || end <= start || end > text.length || pending.length >= 128) return fail();
            pending.push({ key: key || "CONTAINER_VALUE", rawValue: raw, start, end, replacementText: text.slice(start, end), type: classifySecretType(key, raw), placeholder: "", masked });
        };
        if (format === "yaml") yamlCandidates(text, add); else dockerCandidates(text, add);
        pending.sort((a, b) => a.start - b.start); let maskedText = text;
        for (let i = 0; i < pending.length; i++) { const c = pending[i]; if (i && c.start < pending[i - 1].end) return fail(); c.placeholder = `[SOTERAI_PROTECTED_${generation.replace(/-/g, "").toUpperCase()}_${i}]`; if (text.includes(c.placeholder)) return fail(); }
        for (const c of [...pending].reverse()) maskedText = maskedText.slice(0, c.start) + c.masked(c.placeholder) + maskedText.slice(c.end);
        if (pending.some(c => maskedText.includes(c.rawValue) || maskedText.includes(c.replacementText))) return fail(); new SecureEditorModel(maskedText, false);
        if (format === "yaml") yamlDocuments(maskedText); else dockerCandidates(maskedText, () => undefined);
        const candidates = pending.map(({ masked: _privateReplacement, ...candidate }) => candidate);
        return { maskedText, exampleText: "", candidates, ...(format === "yaml" ? { restoreMode: "checkpoint" as const } : {}) };
    } catch { return fail(); }
}
