import { test } from "node:test";
import assert from "node:assert/strict";
const VALUE="TEST_ONLY_F22F_PRIVATE_7c21d89a", GENERATION="26000000-0000-4000-8000-000000000006";
async function module(){const m=await import("../config-scanners/CiSecretPlan");assert.equal(typeof m.createCiSecretPlan,"function");return m}
async function plan(text:string,generation=GENERATION){return(await module()).createCiSecretPlan(text,"ci-yaml",generation)}
const safeError=(e:unknown)=>!String(e).includes(VALUE);
test("F22f admits only declared project GitHub/GitLab CI YAML paths",async()=>{const{ciFormatForPath}=await module();for(const name of[".github/workflows/build.yml","nested/.GITHUB/WORKFLOWS/PUBLIC.YAML",".gitlab-ci.yml",".gitlab/nested/job.yaml"])assert.equal(ciFormatForPath(name),"ci-yaml");for(const name of["build.yml",".github/workflows/nested/job.yml",".circleci/config.yml","Jenkinsfile",".gitlab-ci.json"])assert.equal(ciFormatForPath(name),undefined);for(const name of["../.gitlab-ci.yml","C:/.gitlab-ci.yml","/.gitlab-ci.yml","a\\.gitlab-ci.yml"])assert.throws(()=>ciFormatForPath(name),safeError)});
test("F22f GitHub env strings and multiline private material retain public commands and schema",async()=>{const text='name: public\non: [push]\nenv:\n  API_KEY: '+VALUE+'\n  PRIVATE_KEY: |\n    -----BEGIN PRIVATE KEY-----\n    '+VALUE+'-pem\n    -----END PRIVATE KEY-----\njobs:\n  public:\n    runs-on: ubuntu-latest\n    steps:\n      - run: npm test\n';const result=await plan(text);assert.equal(result.candidates.length,2);assert(!result.maskedText.includes(VALUE));assert.match(result.maskedText,/run: npm test/);assert.equal(result.restoreMode,"checkpoint");assert.equal(result.exampleText,"");assert.equal((await plan(result.maskedText)).candidates.length,0)});
test("F22f GitLab variable value/options inherit only the explicit variable name",async()=>{const text='variables:\n  TOKEN:\n    value: '+VALUE+'\n    options: ['+VALUE+', '+VALUE+'-other]\n    description: Public deployment choice\n    expand: false\nprivate:\n  script: npm test\n';const result=await plan(text);assert.equal(result.candidates.length,3);assert(!result.maskedText.includes(VALUE));assert.match(result.maskedText,/description: Public deployment choice/);assert.match(result.maskedText,/script: npm test/)});
test("F22f compound command arguments/assignments and credential URLs are private whole scalars",async()=>{const command='curl --token '+VALUE+' https://example.invalid/public',url='https://synthetic:'+VALUE+'-url@example.invalid/path',text='jobs:\n  build:\n    steps:\n      - run: '+JSON.stringify(command)+'\n      - run: '+JSON.stringify('export PASSWORD='+VALUE+'-assignment')+'\n      - with:\n          endpoint: '+JSON.stringify(url)+'\n';const result=await plan(text);assert.equal(result.candidates.length,3);assert(result.candidates.some(c=>c.rawValue===command));assert(!result.maskedText.includes(VALUE));assert(!result.maskedText.includes('synthetic:'))});
test("F22f ordinary public CI permissions/job labels/actions and commands stay useful",async()=>{const text='name: Public CI\non: [push, pull_request]\npermissions:\n  contents: read\n  id-token: write\njobs:\n  private:\n    runs-on: ubuntu-latest\n    steps:\n      - uses: actions/checkout@v4\n      - run: npm test\n      - env: {PORT: 3000, MONKEY: banana}\n';const result=await plan(text);assert.equal(result.candidates.length,0);assert.equal(result.maskedText,text)});
test("F22f public names never exempt known credential shapes and credentialed query URLs",async()=>{const secret='ghp_'+'a'.repeat(36),text='name: '+secret+'\nenv:\n  endpoint: https://example.invalid/path?access_token='+VALUE+'\n';const result=await plan(text);assert.equal(result.candidates.length,2);assert(!result.maskedText.includes(secret));assert(!result.maskedText.includes(VALUE))});
test("F22f actual YAML comments preserve public commentary and replace canonical secret spans",async()=>{const result=await plan('# TOKEN='+VALUE+'-comment\nvariables:\n  PASSWORD: '+VALUE+' # ordinary public comment\n');assert.equal(result.candidates.length,2);assert(!result.maskedText.includes(VALUE));assert.match(result.maskedText,/# ordinary public comment/)});
test("F22f exact CI/env references and protected markers are controls while mixed forms withhold",async()=>{const text='env:\n  TOKEN: "${{ secrets.TOKEN }}"\n  PASSWORD: $CI_JOB_TOKEN\n  API_KEY: "${API_KEY}"\n  SECRET: "[SOTERAI_PROTECTED_EXISTING]"\n';assert.equal((await plan(text)).candidates.length,0);await module();for(const value of['prefix${{ secrets.TOKEN }}'+VALUE,'${TOKEN:-'+VALUE+'}','prefix[SOTERAI_PROTECTED_EXISTING]'+VALUE])await assert.rejects(plan('env:\n  TOKEN: '+JSON.stringify(value)+'\n'),safeError)});
test("F22f duplicates, aliases, merges, tags, directives and multiple documents withhold",async()=>{await module();for(const text of['env: {TOKEN: '+VALUE+', TOKEN: duplicate}\n','env: &saved {TOKEN: '+VALUE+'}\ncopy: *saved\n','env: {<<: {TOKEN: '+VALUE+'}}\n','env: {TOKEN: !secret '+VALUE+'}\n','%YAML 1.2\n---\nenv: {TOKEN: '+VALUE+'}\n','env: {TOKEN: '+VALUE+'}\n---\npublic: true\n','env: [unterminated'])await assert.rejects(plan(text),safeError)});
test("F22f unsupported credential collections and scalar types withhold",async()=>{await module();for(const text of['env: {PASSWORD: 123}\n','env: {TOKEN: [opaque]}\n','env: {PASSWORD: {nested: opaque}}\n'])await assert.rejects(plan(text),safeError)});
test("F22f decoded escaped residuals in public scalars/property names withhold complete plans",async()=>{await module();const encoded=VALUE.replaceAll('T','\\u0054');for(const text of['env: {PASSWORD: '+JSON.stringify(VALUE)+'}\ndescription: "prefix'+encoded+'suffix"\n','env: {PASSWORD: '+JSON.stringify(VALUE)+'}\n"'+encoded+'": public\n'])await assert.rejects(plan(text),safeError)});
test("F22f repeated known values under public fields are all replaced and credential names refuse",async()=>{const result=await plan('env: {PASSWORD: '+VALUE+'}\ndescription: '+VALUE+'\n');assert.equal(result.candidates.length,2);assert(!result.maskedText.includes(VALUE));await module();await assert.rejects(plan('"ghp_'+'b'.repeat(36)+'": public\n'),safeError)});
test("F22f CRLF block scalars and inline representations keep exact private spans and valid following fields",async()=>{const text='env:\r\n  PASSWORD: |-\r\n    '+VALUE+'\r\n  PORT: 3000\r\njobs: {}\r\n';const result=await plan(text);assert.equal(result.candidates.length,1);const c=result.candidates[0];assert.equal(text.slice(c.start,c.end),c.replacementText);assert.match(result.maskedText,/PORT: 3000/);assert.equal((await plan(result.maskedText)).candidates.length,0)});
test("F22f text, UTF8/control, nesting, generation and candidate limits fail without values",async()=>{await module();for(const text of['env: {TOKEN: "'+VALUE+'\\0"}\n','env: {TOKEN: "\\uD800"}\n','x'.repeat(65537),'x: '.repeat(70)+'public', 'env:\n'+Array.from({length:129},(_,i)=>'  TOKEN_'+i+': '+VALUE+'-'+i).join('\n')])await assert.rejects(plan(text),safeError);await assert.rejects(plan('env: {TOKEN: '+VALUE+'}\n','invalid'),safeError)});
test("F22f deterministic static YAML corpus preserves public configuration and empty repeated migration",async()=>{for(let i=0;i<64;i++){const text='name: public-'+i+'\nenv:\n  TOKEN: '+VALUE+'-'+i+'\n  PORT: '+(3000+i)+'\njobs: {build: {steps: [{run: npm test}]}}\n',result=await plan(text);assert.equal(result.candidates.length,1);assert(!result.maskedText.includes(VALUE));assert.match(result.maskedText,new RegExp('PORT: '+(3000+i)));assert.equal((await plan(result.maskedText)).candidates.length,0)}});
