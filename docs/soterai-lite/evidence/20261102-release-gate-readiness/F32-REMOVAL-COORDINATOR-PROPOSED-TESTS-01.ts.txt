// Minimal proposed production API:
// export type RemovalAction = the14 fixed IDs in AUTO/MANUAL below;
// export interface RemovalOwner {
//   readonly root:string; readonly local:boolean; readonly trusted:boolean;
//   readonly revision:number; // monotonic owner/session/config/trust epoch
// }
// export interface RemovalDependencies {
//   owner(): RemovalOwner | undefined;
//   probe(id:RemovalAction, owner:RemovalOwner):Promise<unknown>;
//   confirmSummary(items:readonly {id:RemovalAction;status:string}[]):Promise<boolean>;
//   restore(id:RemovalAction):Promise<unknown>; // code-owned map to native reviewed command; NO arguments/grants
//   stopBroker():Promise<unknown>; // await manager.stop(), return manager.ownedSnapshot()
// }
// export class RemovalCoordinator {
//   constructor(deps:RemovalDependencies);
//   run(...untrustedCallerArgs:unknown[]):Promise<RemovalReport>; // arguments ignored
//   dispose():void;
// }
// RemovalReport: {status:'COMPLETE'|'INCOMPLETE'|'CANCELLED'|'BUSY',
//   actions:readonly {id:RemovalAction,status:'RESTORED'|'NOT_APPLICABLE'|
//      'PENDING'|'CANCELLED'|'MANUAL_RECOVERY'}[],
//   broker:'STOPPED'|'PENDING'|'NOT_REQUESTED'}; fixed safe fields/reasons only.
//
// probe ABSENT: {state:'ABSENT'} is an adapter-confirmed absence, never a guess
// from a missing Memento/UNAVAILABLE status. OWNED metadata is
// {state:'OWNED',schemaVersion:1,root,receiptFingerprint:<64 lowercase hex>};
// corrupt/foreign/unreadable/unknown metadata stays PENDING. Adapter probes do
// not decrypt backups or discover/read normal vendor configuration. Dedicated
// profile probes account for retained on-disk recovery, and PSReadLine uses its
// checked metadata journal rather than the obsolete Memento as authority.
// Re-probe exact metadata before a reviewed restore; re-probe ABSENT afterwards.
// Existing per-route native confirmation remains final authority. Restoration
// may remain unavailable with withdrawn flags; coordinator never enables them.
// Broker stop is last, only once no owned/pending/manual route needs recovery.
// COMPLETE covers only the declared known routes, not arbitrary external edits.

const PRIVATE = 'SYNTHETIC_ONLY_F32_PRIVATE_DO_NOT_COPY_e3c91a';

function deferred<T>() {
    let resolve!: (value:T) => void;
    const promise = new Promise<T>(done => { resolve = done; });
    return { promise, resolve };
}
async function bounded<T>(work:Promise<T>):Promise<T> {
    let timer:ReturnType<typeof setTimeout> | undefined;
    try {
        return await Promise.race([work, new Promise<never>((_, reject) => {
            timer = setTimeout(() => reject(Error('Proposed coordinator did not reach the bounded fixture stage.')), 1000);
        })]);
    } finally { if (timer) clearTimeout(timer); }
}
function fixture() {
    let owner:RemovalOwner | undefined = { root:ROOT, local:true, trusted:true, revision:1 };
    const probes = new Map<RemovalAction, unknown>(ALL.map(id => [id, { state:'ABSENT' }]));
    const calls = { probes:[] as RemovalAction[], restores:[] as unknown[][],
        summaries:[] as unknown[], stops:0, forbidden:0 };
    const retained = Object.freeze({ policy:'saved-deny', ciphertext:'opaque-ciphertext', backup:'retained-opaque-backup' });
    const flags = Object.freeze({ preCommit:false, prePush:false, psReadLine:false });
    const hooks = {
        async probe(id:RemovalAction):Promise<unknown> { return probes.get(id); },
        async review():Promise<boolean> { return true; },
        async restore(id:RemovalAction):Promise<unknown> {
            probes.set(id, { state:'ABSENT' }); return { status:id.startsWith('checked') || id === 'secureEditorAssociation' ? 'SAVED' : 'RESTORED' };
        },
        async stop():Promise<unknown> { return { state:'stopped' }; },
    };
    const deps:RemovalDependencies = {
        owner:() => owner,
        async probe(id, captured) {
            assert(ALL.includes(id)); assert.equal(captured.root, owner?.root);
            calls.probes.push(id); return hooks.probe(id);
        },
        async confirmSummary(items) { calls.summaries.push(items); return hooks.review(); },
        async restore(...args:unknown[]) {
            calls.restores.push(args); assert.equal(args.length, 1, 'No approval/path/diff/token arguments may be forwarded.');
            assert(AUTO.includes(args[0] as typeof AUTO[number]), 'Only fixed safe automatic actions may execute.');
            return hooks.restore(args[0] as RemovalAction);
        },
        async stopBroker() { calls.stops++; return hooks.stop(); },
    };
    // No such capability belongs to the pure coordinator. Access is a failure.
    for (const key of ['secrets', 'decrypt', 'policyStore', 'rawBackupRestore', 'configuration', 'executeCommand']) {
        Object.defineProperty(deps, key, { get() { calls.forbidden++; throw Error(PRIVATE); } });
    }
    const coordinator = new RemovalCoordinator(deps);
    return { coordinator, probes, calls, hooks, retained, flags,
        setOwner(next:RemovalOwner | undefined) { owner = next; },
        changeOwner(delta:Partial<RemovalOwner>) { assert(owner); owner = { ...owner, ...delta }; },
        run:(...args:unknown[]) => bounded(coordinator.run(...args)),
    };
}
type F = ReturnType<typeof fixture>;
function status(report:Awaited<ReturnType<RemovalCoordinator['run']>>, id:RemovalAction) {
    const rows = report.actions.filter(row => row.id === id); assert.equal(rows.length, 1); return rows[0].status;
}
function safe(value:unknown) { assert(!JSON.stringify(value).includes(PRIVATE), 'No caller, receipt or exception details may leak.'); }
function untouched(f:F) {
    assert.deepEqual(f.retained, { policy:'saved-deny', ciphertext:'opaque-ciphertext', backup:'retained-opaque-backup' });
    assert.deepEqual(f.flags, { preCommit:false, prePush:false, psReadLine:false }); assert.equal(f.calls.forbidden, 0);
}

test('F32-C10 owned restores run serially and broker stop waits for all completed postconditions', async () => {
    const f = fixture(), entered = deferred<void>(), release = deferred<void>(), trace:string[] = [];
    f.probes.set('checkedPowerShell', owns()); f.probes.set('checkedCmd', owns());
    f.hooks.restore = async id => {
        trace.push('start:' + id);
        if (id === 'checkedPowerShell') { entered.resolve(); await release.promise; }
        f.probes.set(id, { state:'ABSENT' }); trace.push('end:' + id); return { status:'SAVED' };
    };
    f.hooks.stop = async () => { trace.push('broker'); return { state:'stopped' }; };
    const pending = f.run(); await bounded(entered.promise);
    assert.deepEqual(trace, ['start:checkedPowerShell']); release.resolve();
    const result = await pending; assert.equal(result.status, 'COMPLETE');
    assert.deepEqual(trace, ['start:checkedPowerShell', 'end:checkedPowerShell', 'start:checkedCmd', 'end:checkedCmd', 'broker']);
});
