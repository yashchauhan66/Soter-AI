    listeners.push(vscode.workspace.onDidChangeConfiguration(e => { if (e.affectsConfiguration("soterai")) note("configuration-event"); }), context.secrets.onDidChange(e => { if (e.key === KEY) note("hygiene-key-event"); if (e.key === "soterai.vaultKey") note("vault-key-event"); }));
    const store = context.secrets.store; patch(context.secrets, "store", async (key: string, value: string) => { if (key === KEY || key === "soterai.vaultKey") { stores++; assert.equal(Buffer.from(value, "base64").length, 32); } return store.call(context.secrets, key, value); });
    const open = vscode.workspace.openTextDocument; patch(vscode.workspace, "openTextDocument", (...args: unknown[]) => { if (isSource(args[0])) sourceCalls++; const input = args[0] as { content?: string }; if (input?.content) { noValues(input.content); metadataDocuments++; } return Reflect.apply(open, vscode.workspace, args); });

    const show = vscode.window.showTextDocument; patch(vscode.window, "showTextDocument", async (...args: unknown[]) => { const d = args[0] as vscode.TextDocument; noValues(d.getText()); displayed.push(d); return Reflect.apply(show, vscode.window, args); });

        noValues([message, options]); const action = items.find(i => ["Analyze saved project files", "Migrate securely", "Restore raw values", "Create metadata export"].includes(i)); if (!action) return undefined;

        if (hold && action === "Analyze saved project files") { entered = true; return new Promise<string | undefined>(r => finish = r); }

    });
    for (const name of ["showInformationMessage", "showErrorMessage"]) patch(vscode.window, name, async (value: unknown) => { noValues(value); return undefined; });
