import { test } from "node:test";
import assert from "node:assert/strict";
const VALUE = "TEST_ONLY_F22D_PRIVATE_47b2f9a1", GENERATION = "24000000-0000-4000-8000-000000000004";
async function module() { const m = await import("../config-scanners/RegistrySecretPlan"); assert.equal(typeof m.createRegistrySecretPlan, "function"); return m; }
async function plan(text: string, format = "npmrc", generation = GENERATION) { return (await module()).createRegistrySecretPlan(text, format as never, generation); }
const safeError = (e: unknown) => !String(e).includes(VALUE);

test("F22d maps only exact project npmrc/pypirc basenames with bounded canonical relative paths", async () => {
    const { registryFormatForPath } = await module(); for (const [name, format] of [[".npmrc", "npmrc"], ["nested/.NPMRC", "npmrc"], ["config/.pypirc", "pypirc"]]) assert.equal(registryFormatForPath(name), format);
    for (const name of [".npmrc.backup", "npmrc", "a.ini", "package.json", "user.pypirc"]) assert.equal(registryFormatForPath(name), undefined);
    for (const name of ["../.npmrc", "C:/.pypirc", "/.npmrc", "a\\.npmrc", "a//.pypirc"]) assert.throws(() => registryFormatForPath(name), safeError);
});

test("F22d scoped npm tokens and quoted credentials preserve public registry settings and harmless arrays", async () => {
    const text = 'registry=https://registry.example.invalid/\n@public:registry=https://registry.example.invalid/scoped/\n//registry.example.invalid/:_authToken="' + VALUE + '"\npassword=\'' + VALUE + '-quoted\'\nfetch-retries=2\npublic[]=first\npublic[]=second\nMONKEY=banana\n';
    const result = await plan(text); assert.equal(result.candidates.length, 2); assert(!result.maskedText.includes(VALUE)); assert.match(result.maskedText, /registry=https:\/\/registry\.example\.invalid\//); assert.match(result.maskedText, /MONKEY=banana/); assert.match(result.maskedText, /public\[\]=second/); assert.equal(result.restoreMode, "checkpoint"); assert.equal(result.exampleText, "");
});

test("F22d credentialed URLs remove original userinfo while retaining public origin/path and exact private representation", async () => {
    const value = VALUE + "@!", text = 'registry=https://synthetic-user:' + encodeURIComponent(value) + '@registry.example.invalid/path/\n';
    const result = await plan(text); assert.equal(result.candidates.length, 1); assert.equal(result.candidates[0].rawValue, value); assert(!result.maskedText.includes(VALUE)); assert(!result.maskedText.includes("synthetic-user")); const url = new URL(result.maskedText.trim().slice("registry=".length)); assert.equal(url.hostname, "registry.example.invalid"); assert.equal(url.pathname, "/path/"); assert.equal(decodeURIComponent(url.username), result.candidates[0].placeholder); assert.equal(url.password, ""); assert.equal(text.slice(result.candidates[0].start, result.candidates[0].end), result.candidates[0].replacementText);
});

test("F22d npm base64 password and composite Basic auth decode privately and replace with canonical encoded markers", async () => {
    const encoded = Buffer.from(VALUE).toString("base64"), basic = Buffer.from("synthetic:" + VALUE + "-basic").toString("base64"), text = '//registry.example.invalid/:_password=' + encoded + '\n//registry.example.invalid/:_auth=' + basic + '\n';
    const result = await plan(text); assert.equal(result.candidates.length, 2); assert.equal(result.candidates[0].rawValue, VALUE); assert.equal(result.candidates[1].rawValue, "synthetic:" + VALUE + "-basic"); assert(!result.maskedText.includes(encoded)); assert(!result.maskedText.includes(basic)); for (const line of result.maskedText.trim().split("\n")) assert.match(Buffer.from(line.slice(line.indexOf("=") + 1), "base64").toString("utf8"), /^\[SOTERAI_PROTECTED_/); assert.equal((await plan(result.maskedText)).candidates.length, 0);
});

test("F22d pypirc sections/multiline credentials preserve index names and the public token username convention", async () => {
    const text = '[distutils]\nindex-servers =\n    public\n    private\n[public]\nrepository = https://upload.example.invalid/legacy/\nusername = __token__\npassword = ' + VALUE + '-token\n[private]\nrepository = https://private.example.invalid/upload/\nusername = synthetic-private-user\npassword =\n    first\n    ' + VALUE + '-multiline\n';
    const result = await plan(text, "pypirc"); assert.equal(result.candidates.length, 3); assert(!result.maskedText.includes(VALUE)); assert(!result.maskedText.includes("synthetic-private-user")); assert.match(result.maskedText, /username = __token__/); assert.match(result.maskedText, /index-servers =\n    public\n    private/); assert(result.candidates.some(c => c.rawValue === "first\n" + VALUE + "-multiline")); assert.equal(result.restoreMode, "checkpoint");
});

test("F22d actual npm/pypirc comments and npm escapes are handled without treating URLs as comments", async () => {
    const npm = '# API_KEY=' + VALUE + '-comment\npassword=first\\#' + VALUE + '\\;last ; public comment\nregistry=https://registry.example.invalid/\n'; const result = await plan(npm); assert.equal(result.candidates.length, 2); assert(result.candidates.some(c => c.rawValue === "first#" + VALUE + ";last")); assert(!result.maskedText.includes(VALUE)); assert.match(result.maskedText, /; public comment/);
    const pypi = await plan('; TOKEN=' + VALUE + '-comment\n[pypi]\nusername=__token__\npassword="' + VALUE + ' # literal"\n', "pypirc"); assert.equal(pypi.candidates.length, 2); assert(pypi.candidates.some(c => c.rawValue === '"' + VALUE + ' # literal"'));
});

test("F22d bare env references/protected markers remain controls while mixed fallback/interpolation/external references withhold", async () => {
    const text = '_authToken=${NODE_AUTH_TOKEN}\npassword=${PASSWORD?}\nsecret=[SOTERAI_PROTECTED_EXISTING]\nusername=__token__\nMONKEY=banana\n'; const result = await plan(text); assert.equal(result.candidates.length, 0); assert.equal(result.maskedText, text);
    await module(); for (const input of ['_authToken=${TOKEN:-' + VALUE + '}\n', 'password=prefix${TOKEN}\n', 'keyfile=synthetic-no-read.pem\n', 'certfile=/synthetic/no-read.pem\n', '_authToken=prefix[SOTERAI_PROTECTED_EXISTING]' + VALUE + '\n']) await assert.rejects(plan(input), safeError);
    await assert.rejects(plan('[pypi]\npassword=%(other)s\n', "pypirc"), safeError);
});

test("F22d duplicate fields/sections, malformed syntax, npm sections and credential arrays withhold", async () => {
    await module(); for (const text of ['_authToken=' + VALUE + '\n_authToken=duplicate\n', '[registry]\npassword=' + VALUE + '\n', 'password[]=' + VALUE + '\n', 'password="unterminated\n', 'broken line\n']) await assert.rejects(plan(text), safeError);
    for (const text of ['[pypi]\npassword=' + VALUE + '\npassword=duplicate\n', '[pypi]\npassword=' + VALUE + '\n[pypi]\n', '[DEFAULT]\npassword=' + VALUE + '\n', 'password=' + VALUE + '\n', '[broken\npassword=' + VALUE + '\n']) await assert.rejects(plan(text, "pypirc"), safeError);
});

test("F22d noncanonical/binary Basic auth, ambiguous URL credentials and unsupported schemes withhold", async () => {
    await module(); for (const value of ["%%%", "QQ", Buffer.from([0xff, 0]).toString("base64")]) await assert.rejects(plan('_password=' + value + '\n'), safeError); await assert.rejects(plan('_auth=' + Buffer.from(VALUE).toString("base64") + '\n'), safeError);
    for (const value of ['https://user:%ZZ@registry.example.invalid/', 'https://user:' + VALUE + '@registry.example.invalid/?token=opaque', 'ftp://user:' + VALUE + '@registry.example.invalid/']) await assert.rejects(plan('registry=' + value + '\n'), safeError);
});

test("F22d text/candidate/line/UTF8/generation limits fail without value-bearing diagnostics", async () => {
    await module(); for (const text of ['password=' + VALUE + '\0\n', 'password=\ud800\n', 'x'.repeat(65537), '\n'.repeat(4097)]) await assert.rejects(plan(text), safeError); await assert.rejects(plan(Array.from({ length: 129 }, (_, i) => 'password_' + i + '=' + VALUE + '-' + i).join("\n")), safeError); await assert.rejects(plan('password=' + VALUE, "npmrc", "invalid-generation"), safeError);
});

test("F22d residual originals refuse before writes and CRLF original spans remain exact", async () => {
    await module(); await assert.rejects(plan('password=' + VALUE + '\n# unstructured ' + VALUE + '\n'), safeError);
    const text = '[pypi]\r\nusername=__token__\r\npassword=\r\n    first\r\n    ' + VALUE + '\r\n'; const result = await plan(text, "pypirc"); assert.equal(result.candidates.length, 1); assert.equal(result.candidates[0].rawValue, "first\n" + VALUE); assert.equal(text.slice(result.candidates[0].start, result.candidates[0].end), result.candidates[0].replacementText); assert.equal((await plan(result.maskedText, "pypirc")).candidates.length, 0);
});

test("F22d deterministic static npm corpus keeps public settings and makes every replacement idempotent", async () => {
    for (let i = 0; i < 48; i++) { const text = 'registry=https://registry.example.invalid/public' + i + '/\n_authToken=' + VALUE + '-' + i + '\nfetch-retries=' + i + '\n'; const result = await plan(text); assert.equal(result.candidates.length, 1); assert(!result.maskedText.includes(VALUE)); assert.match(result.maskedText, new RegExp('fetch-retries=' + i)); assert.equal((await plan(result.maskedText)).candidates.length, 0); }
});
